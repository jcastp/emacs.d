#+title: Identity — who and where
#+property: header-args:emacs-lisp :tangle "05-identity.el"
#+startup: overview

Who I am in this environment, and where my files live.  Loaded second, right
after =00-core=, because almost everything downstream is a path built from
these.

Neither identity block is committed: the home one loads =personal_info.el= from
Nextcloud, the work one loads =work-info.el= and =company_info.el= from the work
agenda directory.

* Lexical binding
#+begin_src emacs-lisp
  ;; -*- lexical-binding: t; -*-
#+end_src

* Where my files live
Every module builds its paths from these rather than spelling out
=~/Nextcloud/...=.

#+begin_src emacs-lisp
  (defvar my-nextcloud-dir (expand-file-name "Nextcloud/" (getenv "HOME"))
    "Root of my synced Nextcloud directory.")

  (defvar my-agenda-dir (expand-file-name "agenda/" my-nextcloud-dir)
    "Directory holding my agenda and task files.")

  (defvar my-work-dir (expand-file-name "trabajo/agenda/" my-nextcloud-dir)
    "Root of my work directory.  Defined unconditionally because
  `my-work-roam-dir' is built from it in both environments; the work
  identity variables that load uncommitted secrets stay gated below.")

  (defvar my-personal-roam-dir (expand-file-name "org-roam/" my-nextcloud-dir)
    "Directory holding my personal org-roam notes.")

  (defvar my-work-roam-dir (expand-file-name "roam/" my-work-dir)
    "Directory holding my work org-roam notes.")

  (defvar my-roam-dir (if my-workenvironment-p
                          my-work-roam-dir
                        my-personal-roam-dir)
    "The org-roam store active in this environment.
  The two stores are fully independent: separate directories, separate
  databases, no shared nodes and no links between them.")

  (defvar my-escritura-dir (expand-file-name "escritura/" my-nextcloud-dir)
    "Root of my writing directory.")

  (defvar my-blog-dir (expand-file-name "personal/hugo_blog/" my-nextcloud-dir)
    "Root of my ox-hugo blog.")

  (defvar my-agenda-icons-dir (expand-file-name "config/icons/" my-nextcloud-dir)
    "SVG icons shown next to agenda categories.")
#+end_src

* Personal identity
This uses a special file, not commited, with my personal information.
It contains the variables =user-full-name= and =user-mail-address=.

#+begin_src emacs-lisp
  (when my-homeenvironment-p
    ;; to protect my personal information, this file is not commited
    (load (expand-file-name "personal_info.el" my-data-dir))

    ;; where my book list will be stored
    (defvar my-booklist-file (expand-file-name "books.org" my-agenda-dir)
      "My reading list."))
#+end_src

* Work identity and directories
#+begin_src emacs-lisp
  (when my-workenvironment-p
    (defvar my-work-file (concat my-work-dir "work.org")
      "My own work file")

    (defvar my-backlog-file (concat my-work-dir "work_backlog.org")
      "My own work backlog file")

    (defvar my-meetings-file (concat my-work-dir "work_meetings.org")
      "My own work meetings file")

    (defvar my-people-file (concat my-work-dir "work_people.org")
      "Employee 1:1 meeting notes")

    (defvar my-decisions-file (concat my-work-dir "work_decisions.org")
      "Decision log: one entry per decision, enriched at the weekly review.
Deliberately not in `org-agenda-files' -- it is a record, not a task list,
and adding it would push every entry through the `ci' triage view.")

    (defvar my-work-info-file (concat my-work-dir "work-info.el")
      "Personal identity info (user-full-name, user-mail-address) for work.")
    ;; load the information for our company
    (load my-work-info-file)

    ;; this file includes, as of now:
    ;; - my reports (`my/direct-reports')
    (defvar my-company-info-file (concat my-work-dir "company_info.el")
      "Varied information about my company.")
    (load my-company-info-file))
#+end_src
