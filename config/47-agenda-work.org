** Helpers for the planning (W) and review (R) views
The Monday and Friday views need a few things plain tag searches can't express:
date windows ("scheduled up to this Sunday", "before this Monday"), "not
touched for a week", and matches built from this quarter's objective tags.

The objective and goal matches are built from =my/work-objective-tags= and
=my/work-goal-tags= (set in =company_info.el=) /each time a view runs/, so a
task still carrying a retired objective tag drops out of "this quarter"
automatically once the variable changes.  See the work =manual.org= ("The system") for
why horizons are Mondays and why the week is capped at ~15h.

#+begin_src emacs-lisp
  (defvar my/work-week-capacity-hours 15
    "Focus hours available in a typical week (estimated 10-20).
  Used by the THIS WEEK block of the Monday view.")

  ;; ── date bounds (absolute day numbers) ──────────────────────────────
  (defun my/agenda--week-start ()
    "Absolute day of this week's Monday."
    (let* ((today (org-today))
           (dow (calendar-day-of-week (calendar-gregorian-from-absolute today))))
      (- today (mod (- dow 1) 7))))

  (defun my/agenda--week-end ()
    "Absolute day of this week's Sunday."
    (+ (my/agenda--week-start) 6))

  (defun my/agenda--month-end ()
    "Absolute day of the last day of this month."
    (let* ((d (calendar-gregorian-from-absolute (org-today)))
           (m (car d)) (y (nth 2 d)))
      (calendar-absolute-from-gregorian
       (list m (calendar-last-day-of-month m y) y))))

  (defun my/agenda--date (prop)
    "Absolute day of the entry's PROP (\"SCHEDULED\" or \"DEADLINE\"), or nil."
    (let ((ts (org-entry-get nil prop)))
      (and ts (org-time-string-to-absolute ts))))

  (defun my/agenda--repeater-p ()
    "Non-nil if the entry's SCHEDULED timestamp repeats (a ritual)."
    (let ((ts (org-entry-get nil "SCHEDULED")))
      (and ts (string-match-p "[.+]?\\+[0-9]+[hdwmy]" ts))))

  (defun my/agenda--skip ()
    "Position to continue from when skipping the current entry."
    (save-excursion (or (outline-next-heading) (point-max))))

  ;; ── skip functions ──────────────────────────────────────────────────
  (defun my/agenda-skip-unless-dated (prop from to &optional no-repeaters)
    "Keep only entries whose PROP date lies in [FROM, TO] (absolute days).
  FROM or TO may be nil for an open bound.  With NO-REPEATERS, also skip
  entries whose SCHEDULED repeats (the recurring rituals)."
    (let ((d (my/agenda--date prop)))
      (unless (and d
                   (or (null from) (>= d from))
                   (or (null to) (<= d to))
                   (not (and no-repeaters (my/agenda--repeater-p))))
        (my/agenda--skip))))

  (defun my/agenda-skip-if-scheduled ()
    "Skip entries that have a SCHEDULED date: they already have a horizon."
    (when (org-entry-get nil "SCHEDULED")
      (my/agenda--skip)))

  (defun my/agenda-skip-unless-stale (days)
    "Keep only entries with no clock or state change in the last DAYS days.
  Looks at the inactive timestamps between this heading and the next one,
  which is where the LOGBOOK (clock lines and state changes) lives."
    (let* ((limit (- (org-today) days))
           (end (save-excursion (or (outline-next-heading) (point-max))))
           (latest nil))
      (save-excursion
        (while (re-search-forward org-ts-regexp-inactive end t)
          (let ((d (org-time-string-to-absolute (match-string 0))))
            (when (or (null latest) (> d latest)) (setq latest d)))))
      (when (and latest (>= latest limit))
        (my/agenda--skip))))

  ;; ── tag matches from the goal / objective variables ─────────────────
  (defun my/agenda--objectives ()
    (bound-and-true-p my/work-objective-tags))

  (defun my/agenda--not-objectives ()
    "\"-o_a-o_b...\": exclude every current objective tag."
    (mapconcat (lambda (tag) (concat "-" tag)) (my/agenda--objectives) ""))

  (defun my/agenda-match-objectives (&optional todo)
    "Tag match for tasks serving a current-quarter objective, plus TODO part."
    (let ((objs (my/agenda--objectives)))
      (concat (if objs (mapconcat #'identity objs "|")
                "no_objectives_defined")
              (or todo ""))))

  (defun my/agenda-match-goal-not-quarter (&optional todo)
    "Tag match for goal work that serves no current-quarter objective."
    (let ((goals (bound-and-true-p my/work-goal-tags))
          (not-objs (my/agenda--not-objectives)))
      (concat (if goals
                  (mapconcat (lambda (g) (concat "+" g not-objs)) goals "|")
                "no_goals_defined")
              (or todo ""))))

  (defun my/agenda-match-mandatory-a (&optional todo)
    "Tag match for [#A] tasks that carry no current objective tag.
  By the system's rules these are the mandatory ones."
    (concat "+PRIORITY=\"A\"" (my/agenda--not-objectives) (or todo "")))

  ;; ── the week's load ─────────────────────────────────────────────────
  (defun my/week-load ()
    "Return (HOURS COUNT UNESTIMATED) for open tasks scheduled up to Sunday.
  Slipped items (scheduled before this Monday, still open) count on purpose:
  a slipped commitment still costs this week.  Rituals (repeaters) don't."
    (let ((sunday (my/agenda--week-end))
          (minutes 0) (count 0) (unestimated 0))
      (org-map-entries
       (lambda ()
         (let ((d (my/agenda--date "SCHEDULED"))
               (effort (org-entry-get nil "Effort")))
           (when (and d (<= d sunday) (not (my/agenda--repeater-p)))
             (setq count (1+ count))
             (if (and effort (org-duration-p effort))
                 (setq minutes (+ minutes (org-duration-to-minutes effort)))
               (setq unestimated (1+ unestimated))))))
       "/TODO|ONGOING|WAITING" 'agenda)
      (list (/ minutes 60.0) count unestimated)))

  (defun my/week-load-header ()
    "Header for the THIS WEEK block, with the load against capacity."
    (pcase-let ((`(,hours ,count ,unestimated) (my/week-load)))
      (format "[ THIS WEEK — committed: %d items, %.1fh of ~%dh%s%s ]"
              count hours my/work-week-capacity-hours
              (if (> unestimated 0) (format ", %d without Effort" unestimated) "")
              (if (> hours my/work-week-capacity-hours) " — OVER CAPACITY" ""))))

  (defun my/week-load-report ()
    "Echo this week's committed load against `my/work-week-capacity-hours'."
    (interactive)
    (message "%s" (my/week-load-header)))

  ;; ── inbox triage (ci) ───────────────────────────────────────────────
  (defun my/agenda-skip-unless-due-within (days)
    "Keep only entries SCHEDULED or with a DEADLINE up to DAYS from today.
  Past dates count too: a slipped item still needs an estimate."
    (let ((limit (+ (org-today) days))
          (s (my/agenda--date "SCHEDULED"))
          (d (my/agenda--date "DEADLINE")))
      (unless (or (and s (<= s limit)) (and d (<= d limit)))
        (my/agenda--skip))))

  ;; ── OKR alignment (co): one block per current objective ─────────────
  (defun my/agenda--objective-title (tag)
    "Heading of the objective carrying TAG in the * OKR tree of `my-work-file'."
    (or (and (bound-and-true-p my-work-file)
             (file-exists-p my-work-file)
             (with-current-buffer (find-file-noselect my-work-file)
               (org-with-wide-buffer
                (car (org-map-entries (lambda () (org-get-heading t t t t))
                                      (concat "+okr+" tag) 'file)))))
        "objective not found under * OKR"))

  (defun my/agenda--okr-command ()
    "The co view: open work per current objective, then parked goal work.
  An objective whose block is empty has had no work planned against it."
    (list "co" "OKR alignment — this quarter"
          (append
           (mapcar (lambda (tag)
                     `(tags-todo ,(concat "+" tag "/TODO|ONGOING|WAITING")
                                 ((org-agenda-overriding-header
                                   (format "[ %s — %s ]" ,tag (my/agenda--objective-title ,tag)))
                                  (org-agenda-sorting-strategy '(todo-state-down priority-down effort-up)))))
                   (bound-and-true-p my/work-objective-tags))
           '((tags-todo (my/agenda-match-goal-not-quarter "/TODO|ONGOING|WAITING")
                        ((org-agenda-overriding-header "[ GOAL WORK, NOT THIS QUARTER — parked ]")
                         (org-agenda-sorting-strategy '(priority-down effort-up))))))
          '((org-agenda-prefix-format " %i %-16:c %5e "))))

  (defun my/agenda-install-okr-command (&rest _)
    "Rebuild the co entry of `org-agenda-custom-commands' from the objectives.
  Runs before every agenda call, so editing `my/work-objective-tags' takes
  effect without a restart."
    (let ((cmd (my/agenda--okr-command)))
      (setq org-agenda-custom-commands
            (if (assoc "co" org-agenda-custom-commands)
                (mapcar (lambda (c) (if (equal (car c) "co") cmd c))
                        org-agenda-custom-commands)
              (append org-agenda-custom-commands (list cmd))))))

  (advice-add 'org-agenda :before #'my/agenda-install-okr-command)

  ;; super-agenda: an item belongs to "OKR" if it carries a current objective
  (defun my/agenda-item-objective-p (item)
    "Non-nil if agenda ITEM carries one of `my/work-objective-tags'."
    (seq-some (lambda (tag) (member tag (bound-and-true-p my/work-objective-tags)))
              (get-text-property 0 'tags item)))
#+end_src

* Active decisions by tag
Work-only. The saved views above are static; this is not — it prompts for one
or more tags (see "Categorising decisions" in =manual.org=, e.g. =wiz=,
=sensor=, =defend=) and shows only the decisions that carry *all* of them and
are not =superseded=, i.e. the ones still in force. Complements "Work:
Decisions — needs attention" (which is about triage, not product/area
filtering) and the plain =C-c / m= tag search (which does not know about
=STATUS=).

#+begin_src emacs-lisp
  (when my-workenvironment-p

    (defun my/decisions--available-tags ()
      "Return every tag used in `my-decisions-file', read fresh from the buffer."
      (let (tags)
        (with-current-buffer (find-file-noselect my-decisions-file)
          (org-map-entries
           (lambda () (setq tags (append (org-get-tags) tags)))))
        (seq-uniq tags)))

    (defun my/decisions-by-tag (tags)
      "Show active (non-superseded) decisions carrying every tag in TAGS.
Interactively, prompt for one or more tags -- completion offers every tag
actually used in `my-decisions-file' -- then run an `org-ql-search'
restricted to that file: entries must carry ALL of TAGS and must not have
a STATUS property of \"superseded\". A decision still in `Inbox' (no
STATUS yet beyond the fast-capture default of \"accepted\") is treated as
active and included."
      (interactive
       (list (completing-read-multiple "Tags (comma-separated): "
                                        (my/decisions--available-tags)
                                        nil t)))
      (org-ql-search
        (list my-decisions-file)
        `(and (tags-all ,@tags)
              (not (property "STATUS" "superseded")))
        :title (format "Active decisions: %s" (string-join tags "+"))
        :sort '(date)))

    (global-set-key (kbd "C-c q t") #'my/decisions-by-tag))
#+end_src

* Open a direct report (one-key 1:1 prep)
1:1 notes live *per-person* in =my-people-file= (one heading per report, with
that report's running notes and =:people:= action items underneath).  This
command is the low-friction entry point: it prompts for a report (defaulting to
the last one opened, remembered across sessions via savehist), jumps to their
heading — creating it on first use — and narrows to their subtree, so before a
1:1 you have everything about that person in one focused view.

Bound to =<f6>= for a true one-key launch, and to =C-q p= (registered at the end
of this module).

#+begin_src emacs-lisp
  (defvar my/last-report nil
    "Last direct report opened via `my/open-report'.
  Persisted across sessions via `savehist-additional-variables'.")

  (defun my/open-report (person)
    "Open the per-person 1:1 notes for PERSON in `my-people-file'.
  Interactively, prompt among `my/direct-reports', defaulting to the last report
  opened (press RET to reopen it).  Create PERSON's heading if it does not exist
  yet, then narrow to their subtree so the running 1:1 notes and open action
  items sit together in one focused view."
    (interactive
     (let* ((reports (and (boundp 'my/direct-reports) my/direct-reports))
            ;; surface the last-used report first so it is the primary choice
            (ordered (if (and my/last-report (member my/last-report reports))
                         (cons my/last-report (remove my/last-report reports))
                       reports)))
       (list (completing-read (format-prompt "Report" my/last-report)
                              ordered nil nil nil nil my/last-report))))
    (setq my/last-report person)
    (find-file my-people-file)
    (widen)
    (let ((marker (condition-case nil
                      (org-find-olp (list my-people-file person))
                    (error nil))))
      (unless marker
        ;; first 1:1 with this person — create their heading
        (goto-char (point-max))
        (unless (bolp) (insert "\n"))
        (insert (format "* %s\t:%s:\n" person (downcase person)))
        (setq marker (org-find-olp (list my-people-file person))))
      (goto-char marker))
    (org-narrow-to-subtree)
    (if (fboundp 'org-fold-show-subtree)
        (org-fold-show-subtree)
      (org-show-subtree)))

  (keymap-global-set "<f6>" #'my/open-report)
#+end_src
