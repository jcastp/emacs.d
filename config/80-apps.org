* dired config
All the configurations needed for dired.
 #+begin_src emacs-lisp
   (use-package dired
     :ensure nil
     :commands (dired dired-jump)
     ;; Not redundant with `global-hl-line-mode' (10-ui): the local mode is
     ;; sticky, so in a split dired (two-pane copy/move, `dired-dwim-target')
     ;; the line stays highlighted in the window that is not selected.  The
     ;; global mode only highlights the selected window.
     :hook
     (dired-mode . hl-line-mode)
     :config
     (setq dired-recursive-copies 'always)
     (setq dired-recursive-deletes 'always)
     (setq delete-by-moving-to-trash t)
     (setq dired-dwim-target t)
     ;; Revert the directory if there is any change
     (setq dired-auto-revert-buffer 'dired-directory-changed-p)
     ;; directories first in the listings
     (setq dired-listing-switches "-aBhl  --group-directories-first"))
 #+end_src

* Terminal
** eat
A real terminal running zsh inside Emacs.  eat is pure Lisp (nothing to
compile), so every machine gets it, writing laptops included.  See the design
note =20261123-terminal.org= for why eat over vterm.

eat's default input mode binds =C-q= to "send the next key raw", which would
hide =my/key-prefix-map= in every terminal; it moves to =C-c C-q=.  These are
eat's own keymaps, so changing them in =:config= is fine.

Directory tracking and prompt navigation need eat's zsh integration, sourced
at the end of =~/.zshrc= (a no-op outside eat):
=[ -n "$EAT_SHELL_INTEGRATION_DIR" ] && source "$EAT_SHELL_INTEGRATION_DIR/zsh"=
#+begin_src emacs-lisp
  (use-package eat
    :ensure t
    :commands (eat eat-other-window eat-project eat-project-other-window)
    :custom
    (eat-kill-buffer-on-exit t)          ; `exit' closes the buffer
    (eat-enable-yank-to-terminal t)      ; C-y pastes into the shell
    :config
    (define-key eat-semi-char-mode-map (kbd "C-q") nil t)
    (define-key eat-eshell-semi-char-mode-map (kbd "C-q") nil t)
    (define-key eat-semi-char-mode-map (kbd "C-c C-q") #'eat-quoted-input))

  (defun my/eat-new ()
    "Open a new eat terminal, even if one is already running."
    (interactive)
    (eat nil '(4)))

  (let ((term (my/key-define-submap "e" "terminal")))
    (my/key-define "e" "eat"          #'eat         term)
    (my/key-define "n" "new eat"      #'my/eat-new  term)
    (my/key-define "p" "eat (project)" #'eat-project term)
    (my/key-define "s" "eshell"       #'eshell      term))
#+end_src

* Buffer grouping

** ibuffer filter groups
Group buffers by project area. The catch-all "Other" group at the end catches anything unmatched.
#+begin_src emacs-lisp
  (when my-homeenvironment-p
    (setq ibuffer-saved-filter-groups
          `(("personal"
             ;; both ~/.emacs.d and the data dir under Nextcloud/config
             ("Emacs config" (filename . "/.emacs.d/"))
             ("Agenda"       (filename . ,(regexp-quote my-agenda-dir)))
             ("Writing"      (filename . ,(regexp-quote my-escritura-dir)))
             ("Org-roam"     (filename . ,(regexp-quote my-personal-roam-dir)))
             ("Blog"         (filename . ,(regexp-quote
                                           (expand-file-name "personal/" my-nextcloud-dir))))
             ("Special"      (or (name . "^\\*")
                                 (mode  . special-mode))))))

    (add-hook 'ibuffer-mode-hook
              (lambda ()
                (ibuffer-auto-mode 1)
                (ibuffer-switch-to-saved-filter-groups "personal")))
    )

  (when my-workenvironment-p
    (setq ibuffer-saved-filter-groups
          `(("work"
             ("Emacs config" (filename . "/.emacs.d/"))
             ;; Must precede "Work agenda": the roam store is a subdirectory of
             ;; it, and ibuffer takes the first matching group.
             ("Org-roam"     (filename . ,(regexp-quote my-work-roam-dir)))
             ("Work agenda"  (filename . ,(regexp-quote my-work-dir)))
             ("Special"      (or (name . "^\\*")
                                 (mode  . special-mode))))))

    (add-hook 'ibuffer-mode-hook
              (lambda ()
                (ibuffer-auto-mode 1)
                (ibuffer-switch-to-saved-filter-groups "work"))))
#+end_src

* Readers and feeds

** Nov.el
Read e-pub in emacs.

https://tech.toryanderson.com/2022/11/23/viewing-epub-in-emacs/
#+begin_src emacs-lisp
  (when my-homeenvironment-p
    ;; esxml, which nov needs, is installed as nov's dependency and loaded by
    ;; nov itself when the first epub is opened.
    ;;
    ;; `:mode' registers the auto-mode entry without loading nov, which is what
    ;; makes the deferral safe: an `add-to-list' in `:config' would never run.
    ;; Books are read in ETBembo with the prose line spacing, whatever the
    ;; fontaine preset is (at startup it is `hack', a monospace font).  The
    ;; faces are remapped in the book's buffer only, before nov renders it:
    ;; nov fills the text when it renders, in the font in effect then, and
    ;; `nov-mode-hook' would run too late (after the first render).
    (defvar my-nov-reading-face '(:family "ETBembo" :height 150)
      "Face attributes for the text of books in nov.")

    (defvar-local my/nov--reading-face-set nil
      "Non-nil once this nov buffer has its reading face.")

    (defun my/nov-reading-face ()
      "Give this book buffer `my-nov-reading-face' and the prose spacing, once.
  Runs from `nov-pre-html-render-hook', which runs on every render."
      (unless my/nov--reading-face-set
        (setq my/nov--reading-face-set t)
        (apply #'face-remap-add-relative 'variable-pitch my-nov-reading-face)
        ;; `italic' carries the preset's own family, which would otherwise win
        (face-remap-add-relative 'italic
                                 :family (plist-get my-nov-reading-face :family))
        (setq-local line-spacing my-prose-line-spacing)))

    (use-package nov
      :ensure t
      :mode ("\\.epub\\'" . nov-mode)
      :hook (nov-pre-html-render . my/nov-reading-face)
      :config
      ;; 64 columns in ETBembo 150 give the same characters per line as the
      ;; 72 prose columns give in Hack (about 70 on average, measured).
      (setq nov-text-width 64)
      ;; We save the place we left the e-pub in this file:
      (setq nov-save-place-file (expand-file-name "nov-places" my-data-dir)))
    )
#+end_src

** elfeed suite

RSS read in emacs.
#+begin_src emacs-lisp
  (when my-homeenvironment-p
    ;; to store our rss feeds sources
    (use-package elfeed-org
      :ensure t
      :defer t)

    (use-package elfeed
      :ensure t
      :commands (elfeed elfeed-update)
      :init
      ;; Plain settings, safe before the package loads -- and
      ;; `rmh-elfeed-org-files' MUST be set before `elfeed-org' runs.  It used
      ;; to be the other way round: `(elfeed-org)' was called in `:init' while
      ;; the file list was only set in `:config', so elfeed-org started with
      ;; its default list.
      (setq-default elfeed-search-filter "@1-months-ago")
      (setq elfeed-db-directory (expand-file-name "elfeed/db/" my-data-dir))
      (setq rmh-elfeed-org-files
            (list (expand-file-name "elfeed/elfeed.org" my-data-dir)))

      :config
      (require 'elfeed-org)
      (elfeed-org))

    ;; Defined outside the `use-package' so that C-q o f is callable before
    ;; elfeed itself is loaded; calling it autoloads elfeed.
    (defun my/elfeed ()
      "Opens elfeed and updates it in just a go."
      (interactive)
      (elfeed)
      (elfeed-update))
    )
#+end_src

  (let ((online (my/key-define-submap "o" "online")))
    (my/key-define "e" "eww" #'eww online)
    (when my-homeenvironment-p
      (my/key-define "f" "elfeed" #'my/elfeed online)
      (when my-clear-directory-is-mounted-p
        (my/key-define "m" "mastodon" #'mastodon online))))
