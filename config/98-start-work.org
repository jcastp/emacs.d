This will automatically open the relevant file to start working right away:

#+begin_src emacs-lisp
  (defun my/work-initial-setup ()
    "Open my initial view of work: today's roam daily beside `my-work-file'."
    (interactive)
    (org-roam-dailies-find-today)
    (split-window-right)
    (find-file my-work-file))

  (when (file-exists-p my-work-file)
    ;;(find-file my-work-file)
    (my/work-initial-setup)
