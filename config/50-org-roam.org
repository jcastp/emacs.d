org-roam, its UI, the agenda restriction, the random-node openers and the
utility functions -- and the capture templates for both environments.

Personal and work are two fully independent stores: separate directories,
separate databases, no shared nodes and no links between them.  Each
environment's capture templates are self-contained -- neither can see or file
into the other's list.  The active store's directory comes from =my-roam-dir=
in =05-identity.org=, which resolves to =my-personal-roam-dir= or
=my-work-roam-dir= depending on environment.

org-roam talks to its database through emacsql, which uses Emacs's built-in
SQLite when Emacs was compiled with it, and falls back to the =sqlite3= package
otherwise.  A self-built Emacs lacks it when =libsqlite3-dev= was missing at
build time.  That, not the distribution, is what used to break on Debian, so the
test is for the capability itself (and costs no =lsb_release= subprocess).

** Capture templates

Each environment owns its complete capture set.  There is no shared base list
and no =add-to-list=: what you read here is exactly what is offered, and the
two sides are free to drift.  Home files into =personal/= inside the personal
store; work files at the root of the work store, which is itself the
partition.

* Restrict agenda to org-roam
Functions to isolate the org-roam TODOs from the general agenda.
Extracted from https://d12frosted.io/posts/2021-01-16-task-management-with-roam-vol5.html
#+begin_src emacs-lisp
  (setq my/roamtag "roamtag")

  (defun vulpea-buffer-prop-set (name value)
    "Set a file property called NAME to VALUE in buffer file.
    If the property is already set, replace its value."
    (setq name (downcase name))
    (org-with-point-at 1
      (let ((case-fold-search t))
        (if (re-search-forward (concat "^#\\+" name ":\\(.*\\)")
                               (point-max) t)
            (replace-match (concat "#+" name ": " value) 'fixedcase)
          (while (and (not (eobp))
                      (looking-at "^[#:]"))
            (if (save-excursion (end-of-line) (eobp))
                (progn
                  (end-of-line)
                  (insert "\n"))
              (forward-line)
              (beginning-of-line)))
          (insert "#+" name ": " value "\n")))))

  (defun vulpea-buffer-prop-set-list (name values &optional separators)
    "Set a file property called NAME to VALUES in current buffer.
    VALUES are quoted and combined into single string using
    `combine-and-quote-strings'.
    If SEPARATORS is non-nil, it should be a regular expression
    matching text that separates, but is not part of, the substrings.
    If nil it defaults to `split-string-default-separators', normally
    \"[ \f\t\n\r\v]+\", and OMIT-NULLS is forced to t.
    If the property is already set, replace its value."
    (vulpea-buffer-prop-set
     name (combine-and-quote-strings values separators)))

  (defun vulpea-buffer-tags-set (&rest tags)
    "Set TAGS in current buffer, as `:a:b:' -- the form `org-roam-tag-add' uses.
  If filetags value is already set, replace it."
    (vulpea-buffer-prop-set "filetags"
                            (if tags (concat ":" (string-join tags ":") ":") "")))

  (defun vulpea-buffer-prop-get (name)
    "Get a buffer property called NAME as a string."
    (org-with-point-at 1
      (when (re-search-forward (concat "^#\\+" name ": \\(.*\\)")
                               (point-max) t)
        (buffer-substring-no-properties
         (match-beginning 1)
         (match-end 1)))))

  (defun vulpea-buffer-prop-get-list (name &optional separators)
    "Get a buffer property NAME as a list using SEPARATORS.
        If SEPARATORS is non-nil, it should be a regular expression
        matching text that separates, but is not part of, the substrings.
        If nil it defaults to `split-string-default-separators', normally
        \"[ \f\t\n\r\v]+\", and OMIT-NULLS is forced to t."
    (let ((value (vulpea-buffer-prop-get name)))
      (when (and value (not (string-empty-p value)))
        (split-string-and-unquote value separators))))

  (defun vulpea-buffer-tags-get ()
    "Return filetags value in current buffer.
  Tags may be written `:a:b:' or `a b', or a mix of both: splitting on
  spaces alone read `roamtag :a:b:' as two tags, so a `roamtag' inside the
  `:a:b:' part was never seen, never removed, and got a second copy."
    (vulpea-buffer-prop-get-list "filetags" "[ \t:]+"))

  (defun vulpea-project-p ()
    "Return non-nil if current buffer has any todo entry.

            TODO entries marked as done are ignored, meaning the this
            function returns nil if current buffer contains only completed
            tasks."
    (seq-find
     (lambda (type)
       (eq type 'todo))
     (org-element-map
         (org-element-parse-buffer 'headline)
         'headline
       (lambda (h)
         (org-element-property :todo-type h)))))

  (defvar vulpea--updating-tag nil
    "Re-entry guard for `vulpea-project-update-tag'.")

  (defun vulpea-project-update-tag ()
    "Update PROJECT tag in the current buffer."
    (unless vulpea--updating-tag
      (let ((vulpea--updating-tag t))
        (when (and (not (active-minibuffer-window))
                   (vulpea-buffer-p))
          (save-excursion
            (goto-char (point-min))
            (let* ((tags (vulpea-buffer-tags-get))
                   (original-tags tags))
              (if (vulpea-project-p)
                  (setq tags (cons "roamtag" tags))
                (setq tags (remove "roamtag" tags)))
              (setq tags (seq-uniq tags))
              (when (or (seq-difference tags original-tags)
                        (seq-difference original-tags tags))
                (apply #'vulpea-buffer-tags-set tags))))))))

  (defun vulpea-buffer-p ()
    "Return non-nil if the currently visited buffer is a note."
    (and buffer-file-name
         (string-prefix-p
          (expand-file-name (file-name-as-directory org-roam-directory))
          (file-name-directory buffer-file-name))))

  (defun vulpea-project-files ()
    "Return a list of note files containing 'roamtag' tag."
    (seq-uniq
     (seq-map
      #'car
      (org-roam-db-query
       [:select [nodes:file]
                :from tags
                :left-join nodes
                :on (= tags:node-id nodes:id)
                :where (like tag (quote "%\"roamtag\"%"))]))))

  (add-hook 'find-file-hook #'vulpea-project-update-tag)
  (add-hook 'before-save-hook #'vulpea-project-update-tag)
#+end_src

Both hooks stay: at work, =find-file-hook= is what catches a TODO added to a
note elsewhere (another machine, the phone), so it reaches the agenda.

* Seeds: notes waiting to be developed
A note with no content beyond its title is a /seed/: a placeholder made with
=C-c n i= (template =s=) while writing something else.  It carries the =seed=
filetag, kept the same way as =roamtag=: added when the note is empty, removed
the moment it gets a body.  You never set it by hand.  =C-c n S= opens a random
seed; =C-c n f= and typing =seed= lists them.

Only on =before-save-hook=, so opening a note never modifies it.  Dailies are
excluded.  Home only: the work store has no seeds.

#+begin_src emacs-lisp
  (when my-homeenvironment-p
    (defun my/org-roam-seed-p ()
      "Non-nil if this note has nothing but keywords, properties and its title heading."
      (let ((heading (concat "* " (or (vulpea-buffer-prop-get "title") ""))))
        (seq-every-p
         (lambda (line)
           (or (string-match-p "\\`[ \t]*\\([:#].*\\)?\\'" line)
               (string= (string-trim-right line) heading)))
         (split-string (buffer-substring-no-properties (point-min) (point-max))
                       "\n"))))

    (defun my/org-roam-update-seed-tag ()
      "Add the `seed' filetag to an empty note, remove it once the note has a body."
      (when (and (not (active-minibuffer-window))
                 (vulpea-buffer-p)
                 (not (string-prefix-p
                       (expand-file-name org-roam-dailies-directory org-roam-directory)
                       (file-name-directory buffer-file-name))))
        (save-excursion
          (let* ((tags (vulpea-buffer-tags-get))
                 (new (if (my/org-roam-seed-p)
                          (seq-uniq (append tags '("seed")))
                        (remove "seed" tags))))
            (unless (equal tags new)
              (apply #'vulpea-buffer-tags-set new))))))

    (add-hook 'before-save-hook #'my/org-roam-update-seed-tag)

    (defun my/org-roam-tagged-files (tag)
      "Return the files of the org-roam notes tagged TAG."
      (seq-uniq
       (mapcar #'car
               (org-roam-db-query
                [:select [nodes:file] :from tags
                 :left-join nodes :on (= tags:node-id nodes:id)
                 :where (= tags:tag $s1)]
                tag))))

    (defun my/org-roam-random-seed ()
      "Open a random note tagged `seed' -- one waiting to be developed."
      (interactive)
      (if-let* ((files (my/org-roam-tagged-files "seed")))
          (progn (find-file (seq-random-elt files))
                 (message "%d seeds left" (length files)))
        (user-error "No seeds left")))

    (global-set-key (kbd "C-c n S") #'my/org-roam-random-seed))
#+end_src

** Roam TODOs in the work agenda
=org-agenda-files= is a static list, so a TODO living in a roam note is invisible
to every agenda view.  The block above solves half of that: it keeps a =roamtag=
tag on any note in the store that has a live TODO, and =vulpea-project-files= can
list those notes straight from the database.  This is the other half -- the list
is recomputed just before the agenda is built, folding the dynamic set of roam
notes with TODOs into the static set of work files.

Work only.  At home the roam store is a reading and writing corpus whose TODOs
are not agenda material, and the personal agenda stays exactly as it was.

#+begin_src emacs-lisp
  (when my-workenvironment-p

    (defvar my/agenda-files-base org-agenda-files
      "Agenda files that are always present, whatever the roam store holds.
  Captured from `org-agenda-files' as 45-agenda.org left it -- that module
  loads before this one.  This has to be a variable of its own: appending to
  `org-agenda-files' itself would make the list monotonic, so a note whose
  last TODO is done would stay in the agenda for the rest of the session
  instead of dropping out along with its tag.")

    (defvar my/agenda-files-computed nil
      "The list `my/agenda-files-update' last stored in `org-agenda-files'.")

    (defun my/agenda-files-update (&rest _)
      "Fold roam notes that contain a live TODO into `org-agenda-files'.
  Truenames on both sides: the roam database stores paths resolved, so
  without this a file reachable by two paths would be scanned twice and
  each of its headings would show up twice in the agenda.

  Only a list this function owns is rebuilt -- the base list or the one it
  last computed.  A command that let-binds its own `org-agenda-files', such
  as `my/agenda-restrict-to-roam', is left alone: rebuilding would throw its
  restriction away and show the work files instead."
      (when (or (eq org-agenda-files my/agenda-files-base)
                (eq org-agenda-files my/agenda-files-computed))
        (setq org-agenda-files
              (setq my/agenda-files-computed
                    (seq-uniq
                     (append (mapcar #'file-truename my/agenda-files-base)
                             (when (fboundp 'vulpea-project-files)
                               (mapcar #'file-truename (vulpea-project-files)))))))))

    (advice-add 'org-agenda    :before #'my/agenda-files-update)
    (advice-add 'org-todo-list :before #'my/agenda-files-update))
#+end_src

* org-roam random node opening
=C-c n R= opens a random note.  It is org-roam's own =org-roam-node-random=,
restricted to file-level nodes: every note is equally likely, and the headings
that carry an ID (=org-id= adds one whenever a link to a heading is stored) are
left out, so it always opens a whole note.  With a prefix argument it opens in
another window.

=C-c n T= opens a random note that still has a live TODO, which is what the
=roamtag= tag marks (see "Restrict agenda to org-roam" above).  The list comes
from =vulpea-project-files=, one entry per note.

#+begin_src emacs-lisp
  (defun my/random-roam-node (&optional other-window)
    "Open a random org-roam note, each file-level note equally likely.
With prefix argument OTHER-WINDOW, open it in another window."
    (interactive "P")
    (org-roam-node-random other-window
                          (lambda (node) (zerop (org-roam-node-level node)))))

  (defun my/random-roam-todo-node ()
    "Open a random org-roam note that has a live TODO (tagged `roamtag')."
    (interactive)
    (if-let* ((files (vulpea-project-files)))
        (find-file (seq-random-elt files))
      (user-error "No org-roam note has a live TODO")))

  (global-set-key (kbd "C-c n R") #'my/random-roam-node)
  (global-set-key (kbd "C-c n T") #'my/random-roam-todo-node)
#+end_src

* org-roam utility functions

#+begin_src emacs-lisp
  (defun my/org-roam-orphans ()
    "List file-level notes with no links in or out and visit the selected one.
Only file-level nodes are considered (`nodes:level' 0): almost every
sub-heading with its own ID lacks a direct backlink, so including them
would flood the list with headings that live inside well-connected
files.  A note is orphaned only if nothing links to it AND it links to
nothing else; self-links (a note linking to its own ID) do not count
as a connection either way."
    (interactive)
    (let* ((rows (org-roam-db-query
                  [:select [nodes:title nodes:file nodes:id]
                   :from nodes
                   :where (= nodes:level 0)
                   :and (not (in nodes:id
                                 [:select links:dest :from links
                                  :where (<> links:source links:dest)]))
                   :and (not (in nodes:id
                                 [:select links:source :from links
                                  :where (<> links:source links:dest)]))]))
           (candidates (mapcar (lambda (row)
                                 (cons (format "%s (%s)" (nth 0 row)
                                               (file-name-nondirectory (nth 1 row)))
                                       row))
                               rows)))
      (if (not candidates)
          (message "No orphaned notes found")
        (let* ((choice (completing-read
                        (format "Orphaned notes (%d): " (length candidates))
                        (mapcar #'car candidates) nil t))
               (entry (cdr (assoc choice candidates))))
          (find-file (nth 1 entry))))))

  (defun my/org-roam-stale-notes ()
    "List notes not modified in the last 90 days and visit the selected one.
`mtime' lives on the `files' table, not `nodes', so we join and filter in
Elisp using `float-time' to handle org-roam's stored time format."
    (interactive)
    (let* ((cutoff (- (float-time) (* 90 86400.0)))
           (rows (org-roam-db-query
                  [:select [nodes:title nodes:file files:mtime]
                   :from nodes
                   :join files :on (= nodes:file files:file)]))
           (stale (seq-filter
                   (lambda (row) (< (float-time (nth 2 row)) cutoff))
                   rows)))
      (if (not stale)
          (message "No stale notes found")
        (let* ((choice (completing-read
                        (format "Stale notes (%d): " (length stale))
                        (mapcar #'car stale) nil t))
               (entry (cl-find choice stale :key #'car :test #'string=)))
          (find-file (cadr entry))))))

  (defun my/org-roam-creative-prompt ()
    "Open two random notes side by side and ask how they connect.
Only distinct, file-level, non-daily notes.  The window layout is not
cleared first: at work that is the side-by-side view `99-start-work' opens."
    (interactive)
    (let* ((daily (expand-file-name org-roam-dailies-directory org-roam-directory))
           (nodes (seq-filter
                   (lambda (n) (and (zerop (org-roam-node-level n))
                                    (not (string-prefix-p daily (org-roam-node-file n)))))
                   (org-roam-node-list)))
           (a (seq-random-elt nodes))
           (b (seq-random-elt (remove a nodes))))

      (org-roam-node-visit b t)
      (message "How does '%s' connect to '%s'?"
               (org-roam-node-title a) (org-roam-node-title b))))
